**JARVIS OS** replaces the traditional Linux desktop (bars, docks, icons, overlapping windows) with a **standalone Arch Linux Wayland display session** powered by **Minimal Hyprland 0.57+ (`hyprland.lua`)** as the display server, **Quickshell 0.3.x (Qt6/QML)** as the dynamic widget HUD, and your existing Python voice/agent stack (**Cactus-Needle** offline router + **Gemini Live API** + **Hermes** orchestrator), tuned strictly for **8 GB RAM**.

---

## 1. System Architecture & Spatial Model

Every visual element and OS process maps onto **four vertical layers** and **three Wayland depth planes**:

| Layer / Plane | Component | Responsibility |
| --- | --- | --- |
| **Hardware & Kernel** | Arch Linux + `linux-cachyos` | Low-latency BORE/sched-ext scheduling for PipeWire echo cancellation, Silero VAD, and sub-200 ms local voice routing.

 |
| **Display Server** | Minimal Hyprland 0.57+ (`jarvis-minimal.lua`) | Owns GPU (DRM/KMS), `libinput` mouse/keyboard, rootless XWayland, GPU frosted-glass blur, and floating app rules with `misc.vfr = true`. |
| **Plane 1 (Background)** | `CoreCanvas.qml` (`WlrLayer.Background`) | Fullscreen `#05080F` dark canvas + central Arc Reactor voice core (35%–65% width) driven by PipeWire audio peaks, with **0% active timers in standby**.

 |
| **Plane 2 (Widget HUD)** | `WidgetCanvas.qml` (`WlrLayer.Top` / `Overlay`) | Click-through holographic cards on **Left Rail** (2%–32%, max 3) and **Right Stage** (68%–98%, max 2), 5-widget cap, `TopBar.qml` overflow pills, and `Loader`/`LazyLoader` RAM unloading.

 |
| **Plane 3 (App Stage)** | Hyprland Floating Panels + `special:vault` | Real Linux GUI apps (browser, terminal, video, IDE) spawn borderless at `53% 12%` (`44% 76%`) beside the core and park invisibly in `special:vault` on `"Clear HUD"`.

 |
| **Voice & Agent Plane** | `voice_layer/` + `widget_engine/` + `orchestration/` | Offline Cactus-Needle (<200 ms) controls HUD/Hyprland directly; Gemini Live + Hermes stream structured NDJSON widget events over a Unix socket.

 |


## 5. Phase 4 — Bidirectional NDJSON Protocol & Python HUD Bridge

Use **two complementary IPC channels**:

1. **`qs ipc call jarvis ...`** — zero-setup CLI triggers for instant offline Cactus-Needle commands (`setState`, `clearHud`, `pinLast`, `expandLast`, `togglePalette`).
2. **Unix Domain Socket (`/run/user/$UID/jarvis-hud.sock`)** — full-duplex newline-delimited JSON (NDJSON) streaming rich widget payloads from Python to Quickshell, and returning button clicks/approvals/text commands from Quickshell back to Python.

### 5.1 `widget_engine/protocol.py`

```python
from __future__ import annotations
import json
from typing import Any, Literal
from pydantic import BaseModel, Field

HudAction = Literal[
    "spawn_widget",
    "update_widget",
    "pin_widget",
    "dismiss_widget",
    "expand_widget",
    "set_voice_state",
    "set_transcript",
]

WidgetZone = Literal["left", "right", "modal"]
VoiceState = Literal["standby", "listening", "thinking", "executing", "speaking"]


class HudCommand(BaseModel):
    action: HudAction
    id: str = ""
    type: str = ""
    zone: WidgetZone = "right"
    ttl: int = 25  # -1 = pinned (never auto-dismiss)
    title: str = ""
    expand_cmd: str = ""
    state: VoiceState | None = None
    text: str = ""
    payload: dict[str, Any] = Field(default_factory=dict)

    def to_ndjson(self) -> bytes:
        return (self.model_dump_json(exclude_none=True) + "\n").encode("utf-8")


class WidgetReturnEvent(BaseModel):
    event: Literal["widget_action"] = "widget_action"
    id: str
    action_name: str  # approve | reject | select_item | toggle_task | submit_command | expand | dismiss
    data: dict[str, Any] = Field(default_factory=dict)

    @classmethod
    def from_line(cls, raw: bytes) -> "WidgetReturnEvent":
        return cls.model_validate(json.loads(raw.decode("utf-8")))

```

### 5.2 `widget_engine/hud_bridge.py`

```python
from __future__ import annotations
import asyncio
import os
from collections.abc import Awaitable, Callable
from pathlib import Path
from .protocol import HudCommand, WidgetReturnEvent

ActionCallback = Callable[[WidgetReturnEvent], Awaitable[None]]


def default_socket_path() -> str:
    runtime_dir = os.environ.get("XDG_RUNTIME_DIR", f"/run/user/{os.getuid()}")
    return str(Path(runtime_dir) / "jarvis-hud.sock")


class HudBridge:
    def __init__(self, socket_path: str | None = None) -> None:
        self.socket_path = socket_path or default_socket_path()
        self._reader: asyncio.StreamReader | None = None
        self._writer: asyncio.StreamWriter | None = None
        self._callbacks: list[ActionCallback] = []
        self._listen_task: asyncio.Task[None] | None = None
        self._lock = asyncio.Lock()

    def on_action(self, cb: ActionCallback) -> None:
        self._callbacks.append(cb)

    async def connect(self) -> None:
        async with self._lock:
            if self._writer and not self._writer.is_closing():
                return
            self._reader, self._writer = await asyncio.open_unix_connection(self.socket_path)
            self._listen_task = asyncio.create_task(self._read_loop())

    async def send(self, cmd: HudCommand) -> None:
        for attempt in range(3):
            try:
                await self.connect()
                assert self._writer is not None
                self._writer.write(cmd.to_ndjson())
                await self._writer.drain()
                return
            except (ConnectionError, FileNotFoundError):
                await self.close()
                if attempt == 2:
                    raise
                await asyncio.sleep(0.25 * (attempt + 1))

    async def _read_loop(self) -> None:
        assert self._reader is not None
        try:
            while line := await self._reader.readline():
                event = WidgetReturnEvent.from_line(line)
                for cb in self._callbacks:
                    asyncio.create_task(cb(event))
        except asyncio.CancelledError:
            pass
        finally:
            await self.close()

    async def close(self) -> None:
        if self._listen_task:
            self._listen_task.cancel()
            self._listen_task = None
        if self._writer:
            self._writer.close()
            try:
                await self._writer.wait_closed()
            except Exception:
                pass
        self._reader = None
        self._writer = None

```

### 6.3 `ui/jarvis_shell/StateStore.qml`

Manages the **5-widget cap** (max 3 left, max 2 right), **overflow pill collapsing**, **IPC handler** (with explicit Quickshell 0.3 types), and the **bidirectional Unix `SocketServer**`:

```qml
pragma Singleton
import QtQuick
import Quickshell
import Quickshell.Io
import Quickshell.Hyprland

Singleton {
    id: root

    property string voiceState: "standby" // standby | listening | thinking | executing | speaking
    property string transcript: ""
    property var activeWidgets: []        // [{id, type, zone, ttl, title, expand_cmd, payload, pinned}]
    property var minimizedPills: []       // Overflow or collapsed widgets
    property var activeModal: null        // {id, type, title, payload}
    property bool paletteOpen: false

    readonly property int maxTotal: 5
    readonly property int maxLeft: 3
    readonly property int maxRight: 2
    readonly property string socketPath: (Quickshell.env("XDG_RUNTIME_DIR") || "/tmp") + "/jarvis-hud.sock"

    // Connected client socket reference for returning UI actions to Python
    property var _clientSocket: null

    function handleCommand(msg) {
        switch (msg.action) {
        case "set_voice_state":
            voiceState = msg.state || "standby";
            if (voiceState === "standby") transcript = "";
            break;
        case "set_transcript":
            transcript = msg.text || "";
            break;
        case "spawn_widget":
            spawnWidget(msg);
            break;
        case "update_widget":
            updateWidget(msg);
            break;
        case "pin_widget":
            setPinned(msg.id, true);
            break;
        case "dismiss_widget":
            if (msg.id === "all") clearUnpinned();
            else dismissWidget(msg.id);
            break;
        case "expand_widget":
            expandWidget(msg.id);
            break;
        }
    }

    function spawnWidget(w) {
        if (w.zone === "modal") {
            activeModal = w;
            return;
        }
        const zone = (w.zone === "left") ? "left" : "right";
        const item = {
            id: w.id || ("w_" + Date.now()),
            type: w.type || "web_intel",
            zone: zone,
            ttl: (w.ttl !== undefined) ? w.ttl : 25,
            title: w.title || w.type,
            expand_cmd: w.expand_cmd || "",
            payload: w.payload || {},
            pinned: (w.ttl === -1)
        };

        let list = activeWidgets.filter(x => x.id !== item.id);
        let pills = minimizedPills.filter(x => x.id !== item.id);

        // Enforce per-zone and global 5-widget cap by collapsing oldest unpinned card
        const zoneCap = (zone === "left") ? maxLeft : maxRight;
        const zoneCount = () => list.filter(x => x.zone === zone).length;

        while (zoneCount() >= zoneCap || list.length >= maxTotal) {
            let victimIdx = list.findIndex(x => x.zone === zone && !x.pinned);
            if (victimIdx < 0) victimIdx = list.findIndex(x => !x.pinned);
            if (victimIdx < 0) break;
            pills.push(list[victimIdx]);
            list.splice(victimIdx, 1);
        }

        list.push(item);
        minimizedPills = pills;
        activeWidgets = list;
    }

    function updateWidget(w) {
        activeWidgets = activeWidgets.map(x =>
            x.id === w.id ? Object.assign({}, x, {
                title: w.title || x.title,
                payload: Object.assign({}, x.payload, w.payload || {})
            }) : x
        );
    }

    function setPinned(id, pinned) {
        activeWidgets = activeWidgets.map(x =>
            x.id === id ? Object.assign({}, x, { pinned: pinned, ttl: pinned ? -1 : 25 }) : x
        );
    }

    function dismissWidget(id) {
        if (activeModal && activeModal.id === id) {
            activeModal = null;
            return;
        }
        activeWidgets = activeWidgets.filter(x => x.id !== id);
        minimizedPills = minimizedPills.filter(x => x.id !== id);
    }

    function minimizeWidget(id) {
        const target = activeWidgets.find(x => x.id === id);
        if (!target) return;
        activeWidgets = activeWidgets.filter(x => x.id !== id);
        minimizedPills = minimizedPills.concat([target]);
    }

    function restorePill(id) {
        const target = minimizedPills.find(x => x.id === id);
        if (!target) return;
        spawnWidget(target);
    }

    function clearUnpinned() {
        activeWidgets = activeWidgets.filter(x => x.pinned);
        minimizedPills = [];
    }

    function expandWidget(id) {
        const target = activeWidgets.find(x => x.id === id) || activeWidgets[activeWidgets.length - 1];
        if (!target || !target.expand_cmd) return;
        Hyprland.dispatch("exec " + target.expand_cmd);
        minimizeWidget(target.id);
        sendAction(target.id, "expand", { cmd: target.expand_cmd });
    }

    function sendAction(id, actionName, data) {
        if (!_clientSocket || !_clientSocket.connected) return;
        const line = JSON.stringify({
            event: "widget_action",
            id: id,
            action_name: actionName,
            data: data || {}
        }) + "\n";
        _clientSocket.write(line);
        _clientSocket.flush();
    }

    // Channel 1: Fast CLI IPC for Cactus-Needle (<200ms offline commands)
    IpcHandler {
        target: "jarvis"
        function setState(state: string): void { root.voiceState = state; }
        function clearHud(): void { root.clearUnpinned(); }
        function pinLast(): void {
            if (root.activeWidgets.length > 0) {
                root.setPinned(root.activeWidgets[root.activeWidgets.length - 1].id, true);
            }
        }
        function expandLast(): void {
            if (root.activeWidgets.length > 0) {
                root.expandWidget(root.activeWidgets[root.activeWidgets.length - 1].id);
            }
        }
        function togglePalette(): void { root.paletteOpen = !root.paletteOpen; }
        function dismissId(id: string): void { root.dismissWidget(id); }
    }

    // Channel 2: Full-Duplex Unix SocketServer for Python Backend
    SocketServer {
        path: root.socketPath
        active: true
        handler: Socket {
            id: clientSock
            onConnectedChanged: {
                if (connected) root._clientSocket = clientSock;
            }
            parser: SplitParser {
                onRead: msg => {
                    try { root.handleCommand(JSON.parse(msg)); }
                    catch (e) { console.warn("Invalid HUD JSON:", e); }
                }
            }
        }
    }
}

```
